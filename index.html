Overtime is weekly, not per pay period, and that distinction is someone's paycheck

A two-week pay period totalling 80 hours looks clean until you see it was 52 and 28. The split runs per week against the threshold, so the first week owes 12 hours of overtime that a period-level total would hide entirely.

/** Great-circle distance in metres between two points (haversine). */
export function distanceMeters(aLat: number, aLng: number, bLat: number, bLng: number): number {
  const R = 6_371_000;
  const toRad = (d: number) => (d * Math.PI) / 180;
  const dLat = toRad(bLat - aLat), dLng = toRad(bLng - aLng);
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(aLat)) * Math.cos(toRad(bLat)) * Math.sin(dLng / 2) ** 2;
  return Math.round(2 * R * Math.asin(Math.min(1, Math.sqrt(h))));
}

app/packages/shared/src/payroll.ts

/** Split worked minutes into regular and overtime per week: anything above
 *  `overtimeWeeklyHours` in one week is overtime. */
export function splitOvertime(minutesByWeek: Record<string, number>, overtimeWeeklyHours: number): OvertimeSplit {
  const threshold = Math.max(0, overtimeWeeklyHours) * 60;
  const weeks: OvertimeWeek[] = Object.keys(minutesByWeek).sort().map((weekStart) => {
    const minutes = Math.max(0, Math.round(minutesByWeek[weekStart] ?? 0));
    const regularMinutes = Math.min(minutes, threshold);
    return { weekStart, minutes, regularMinutes, overtimeMinutes: minutes - regularMinutes };
  });
  return {
    regularMinutes: weeks.reduce((s, w) => s + w.regularMinutes, 0),
    overtimeMinutes: weeks.reduce((s, w) => s + w.overtimeMinutes, 0),
    weeks,
  };
}

/** Minnesota visit edit reason codes (HHAeXchange MN spec) that fit each kind
 *  of flag, and the action a coordinator records. */
function reasonFor(flags: EvvFlag[]): { reasonCode: string; reasonNote: string; actionCode: string } {
  if (flags.includes("manual_entry")) return { reasonCode: "221", reasonNote: "Times taken from the paper log and confirmed with the person.", actionCode: "125" };
  if (flags.includes("out_of_range_in") || flags.includes("out_of_range_out")) return { reasonCode: "204", reasonNote: "Community outing; service started at the day program.", actionCode: "123" };
  if (flags.includes("no_gps_in") || flags.includes("no_gps_out")) return { reasonCode: "218", reasonNote: "No signal in the basement; clocked as soon as the app allowed.", actionCode: "123" };
  if (flags.includes("unscheduled")) return { reasonCode: "222", reasonNote: "Picked up the shift by phone; the schedule was not updated.", actionCode: "111" };
  if (flags.includes("missing_clock_out")) return { reasonCode: "209", reasonNote: "Forgot to clock out at the end of the shift.", actionCode: "123" };
  if (flags.includes("late_clock_in")) return { reasonCode: "211", reasonNote: "Forgot to clock in on arrival.", actionCode: "123" };
  return { reasonCode: "222", reasonNote: "See the shift notes.", actionCode: "123" };
}

Four candidate matches for one ASIN. Only one survives.

Toggle any attribute and watch the winner change. The ordering is the SQL window function in the Source tab, run in the browser: a human's manual match outranks an image-verified one, which outranks an unreviewed one, which outranks a rejected one; in stock beats out of stock; profit only breaks a tie. The losing rows are deleted, so a buyer never sees two prices for the same ASIN.

packages/backend/src/jobs/scanProcessor.ts

-- One scraped product can match the same ASIN several ways. Keep the single
-- best row per ASIN and delete the rest: a human's manual match outranks an
-- image-verified one, which outranks an unreviewed one, which outranks one a
-- reviewer rejected; in-stock beats out-of-stock; profit breaks the tie.
delete from matched_products mp
using (
  select mp2.id, row_number() over (
    partition by al2.asin
    order by (mp2.match_method = 'manual')            desc,
             (mp2.image_verified is true)             desc,
             (mp2.image_verified is distinct from false) desc,
             sp2.in_stock                             desc,
             mp2.net_profit                           desc nulls last
  ) rn
  from matched_products mp2
  join scraped_products sp2 on sp2.id = mp2.scraped_product_id
  join amazon_listings  al2 on al2.id = mp2.amazon_listing_id
  where mp2.is_bundle = false and sp2.marketplace_id = ?
) ranked
where mp.id = ranked.id and ranked.rn > 1

src/categorize/engine.ts

/** Below this, we ask rather than assume. */
export const CONFIDENT = 0.8;

/**
 * Run the rule pack over everything not yet categorised. A confident rule hit
 * becomes 'auto' and is ready to post; anything else becomes 'needs_client'
 * and shows up in the question list. Nothing is ever silently guessed.
 */
export function categorizePending(db: DB): CategorizeSummary {
  const rules = listRules(db);
  const pending = db
    .prepare(`SELECT * FROM bank_transactions WHERE status = 'uncategorized' ORDER BY txn_date, id`)
    .all() as BankTxn[];

  for (const txn of pending) {
    const rule: Rule | null = findRule(db, txn, rules);
    if (rule && rule.confidence >= CONFIDENT) {
      update.run('auto', rule.account_id, rule.confidence, rule.id, txn.id);
      auto++;
    } else {
      update.run('needs_client', rule ? rule.account_id : ask.id, rule ? rule.confidence : 0, rule?.id ?? null, txn.id);
      needsClient++;
    }
  }
}
